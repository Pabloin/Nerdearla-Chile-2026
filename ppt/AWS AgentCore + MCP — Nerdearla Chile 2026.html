STEP 1 — MCP SERVER (Tool para CloudWatch)
# mcp_cloudwatch_server.py
from mcp.server.fastmcp import FastMCP
import boto3

mcp = FastMCP("cloudwatch-tools")
cw = boto3.client("cloudwatch")

@mcp.tool()
async def get_metric_alarm_status(
    service_name: str
) -> dict:
    """Obtiene el estado de alarmas para un servicio."""
    response = cw.describe_alarms(
        AlarmNamePrefix=service_name,
        StateValue="ALARM"
    )
    return {
        "alarms": response["MetricAlarms"],
        "count": len(response["MetricAlarms"])
    }

# Deployar en AgentCore Runtime:
# agentcore configure -e mcp_cloudwatch_server.py \
#   --protocol MCP --name cloudwatch-mcp
# agentcore launch
STEP 2 — AGENTCORE GATEWAY (Registrar MCP Targets)
# setup_gateway.py
import boto3

client = boto3.client("bedrock-agentcore-control")

# 1. Crear el Gateway
gateway = client.create_gateway(
    name="devops-gateway",
    authorizerType="CUSTOM_JWT",
    authorizerConfiguration={
        "customJWTAuthorizer": {
            "allowedClients": [COGNITO_CLIENT_ID],
            "discoveryUrl": COGNITO_DISCOVERY_URL
        }
    }
)

# 2. Agregar MCP Server como target
target = client.create_gateway_target(
    gatewayIdentifier=gateway["gatewayId"],
    name="cloudwatch-mcp-target",
    targetConfiguration={
        "mcp": {
            "endpoint": {
                "url": AGENTCORE_RUNTIME_URL
            }
        }
    }
)
STEP 3 — ORCHESTRATOR AGENT (Strands + AgentCore)
# orchestrator_agent.py
from strands import Agent
from strands.tools.mcp import MCPClient
from bedrock_agentcore.runtime import BedrockAgentCoreApp

app = BedrockAgentCoreApp()

# Conectar al Gateway como MCP client
mcp_client = MCPClient(
    endpoint=GATEWAY_ENDPOINT,
    token=ACCESS_TOKEN
)

agent = Agent(
    model="us.amazon.nova-pro-v1:0",
    system_prompt="""Eres un experto DevOps.
    Cuando detectes problemas, usa las herramientas 
    MCP para diagnosticar y resolver. Siempre 
    explica tus acciones en español.""",
    tools=mcp_client.list_tools()
)

@app.entrypoint
async def handle(payload: dict):
    result = await agent.invoke_async(
        payload["user_message"]
    )
    return {"response": result.message}
STEP 4 — DEMO CLIENT (Invoke the Agent)
# demo_client.py — lo que corres en vivo
import boto3
import json

runtime = boto3.client("bedrock-agentcore")

# 🔴 LIVE DEMO — ejecutar esto en pantalla
response = runtime.invoke_agent_runtime(
    agentRuntimeArn=AGENT_ARN,
    sessionId="nerdearla-demo-01",
    payload=json.dumps({
        "user_message": "Hola! Nuestro servicio \
            'checkout-api' está dando errores \
            500 desde hace 10 minutos. \
            Puedes investigar y ayudarme?"
    })
)

# Output en tiempo real (streaming)
for chunk in response["body"]:
    print(chunk["chunk"]["bytes"].decode(), 
          end="", flush=True)

Pro Tips 🎤

🧪

Pre-deploy todo

Haz deploy de los agentes y MCP servers el día antes. En el demo, solo muestra el invoke y el resultado.

🔴

Plan B siempre

Graba una corrida exitosa. Si algo falla en vivo, muestra el video. El público lo entenderá.

🌎

Use case local

Menciona un caso real latinoamericano: fintech, retail, logística. El público se conecta más.

📊

Muestra Observability

Abre CloudWatch Traces en vivo para mostrar cómo cada tool call es trazado. Muy impactante visualmente.

🔗

QR Code al repo

Termina con un QR al GitHub repo con todo el código. La comunidad lo agradece muchísimo.

💬

Habla en español

Los comentarios en el código en español, el scenario en español. Hace el demo mucho más cercano.
