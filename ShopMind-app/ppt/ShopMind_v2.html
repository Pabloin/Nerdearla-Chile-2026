supervisor_agent.py — El cerebro
# Supervisor Agent conecta al MCP server
from strands import Agent
from strands.tools.mcp import MCPClient
from bedrock_agentcore.runtime import \
    BedrockAgentCoreApp

app = BedrockAgentCoreApp()

# Conecta al MCP server local
mcp_client = MCPClient(
  lambda: streamablehttp_client(
    "http://localhost:8001/mcp"
  )
)

agent = Agent(
  model="us.amazon.nova-pro-v1:0",
  system_prompt="""Eres ShopMind.
  Tus herramientas buscan productos
  REALES en MercadoLibre Chile.
  
  Muestra siempre los URLs completos
  para que el usuario pueda hacer clic.
  
  Responde en español.""",
  tools=[mcp_client]  # MCP tools
)

@app.entrypoint
async def handle(payload):
  async for chunk in \
    agent.stream_async(
      payload["message"],
      session_id=payload.get("session_id")
    ):
    yield chunk
mcp_server.py — 4 tools en 1 servidor
# FastMCP server con 4 tools
from mcp.server.fastmcp import FastMCP
import subprocess, json

mcp = FastMCP("shopmind-tools")

def _scrape(query, limit=10):
  """Llama al scraper de Puppeteer"""
  result = subprocess.run(
    ["node", "scraper.js", query, str(limit)],
    capture_output=True, text=True
  )
  return json.loads(result.stdout)

@mcp.tool()
async def web_search(query: str) -> str:
  """Busca productos en MercadoLibre"""
  products = _scrape(query)
  return json.dumps(products)

@mcp.tool()
async def price_compare(
  product_name: str,
  max_price_usd: float
) -> str:
  """Filtra por presupuesto"""
  products = _scrape(product_name)
  filtered = [p for p in products
    if p["price_usd"] <= max_price_usd]
  return json.dumps(filtered)

scraper.js — Puppeteer scraper
// Scraper de MercadoLibre Chile
const puppeteer = require('puppeteer');

(async () => {
  const query = process.argv[2];
  const limit = process.argv[3] || 10;
  
  const browser = await puppeteer.launch({
    headless: true
  });
  
  const page = await browser.newPage();
  
  // Busca en MercadoLibre Chile
  await page.goto(
    `https://listado.mercadolibre.cl/${query}`
  );
  
  // Extrae productos
  const products = await page.evaluate(() => {
    return Array.from(
      document.querySelectorAll('.ui-search-result')
    ).map(el => ({
      title: el.querySelector('.ui-search-item__title')
        ?.textContent,
      price_clp: parseFloat(
        el.querySelector('.price-tag-amount')
          ?.textContent.replace(/\D/g, '')
      ),
      url: el.querySelector('a')?.href,
      rating: parseFloat(
        el.querySelector('.ui-search-reviews__rating')
          ?.textContent
      )
    }));
  });
  
  console.log(JSON.stringify(products.slice(0, limit)));
  await browser.close();
})();
