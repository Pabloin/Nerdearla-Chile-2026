// 04 — Código del Demo

Key Code Snippets

supervisor_agent.py — El cerebro
# Agents-as-Tools pattern: sub-agentes
# son herramientas del supervisor
from strands import Agent
from strands.tools.mcp import MCPClient
from bedrock_agentcore.runtime import \
    BedrockAgentCoreApp

app = BedrockAgentCoreApp()
mcp = MCPClient(GATEWAY_URL, token=TOKEN)

supervisor = Agent(
  model="us.amazon.nova-pro-v1:0",
  system_prompt="""Eres ShopMind.
  Llama a search, price, review y
  budget agents EN PARALELO.
  Responde en español, claro y
  con emojis.""",
  # sub-agentes como tools
  tools=[search_agent, price_agent,
         review_agent, budget_agent]
)

@app.entrypoint
async def handle(payload):
  async for chunk in \
    supervisor.stream_async(
      payload["message"]
    ):
    yield chunk
mcp_price_server.py — MCP Tool
# FastMCP server deployado en
# AgentCore Runtime via:
# agentcore configure -e server.py \
#   --protocol MCP && agentcore launch
from mcp.server.fastmcp import FastMCP
import boto3, json

mcp = FastMCP("shopmind-price-tools")

@mcp.tool()
async def price_compare(
  product_name: str,
  max_price_usd: float
) -> list[dict]:
  """Compara precios en tiendas."""
  res = boto3.client("lambda").invoke(
    FunctionName="shopmind-scraper",
    Payload=json.dumps({
      "query": product_name,
      "max": max_price_usd
    })
  )
  return json.loads(
    res["Payload"].read()
  )
setup_gateway.py — Registra MCP targets
import boto3

client = boto3.client(
  "bedrock-agentcore-control"
)

# 1. Crear el Gateway
gw = client.create_gateway(
  name="shopmind-gateway",
  authorizerType="CUSTOM_JWT",
  authorizerConfiguration={
    "customJWTAuthorizer": {
      "allowedClients": [CLIENT_ID],
      "discoveryUrl":   COGNITO_URL
    }
  }
)

# 2. Registrar cada MCP server
for name, url in MCP_SERVERS.items():
  client.create_gateway_target(
    gatewayIdentifier=gw["gatewayId"],
    name=f"shopmind-{name}",
    targetConfiguration={
      "mcp": {
        "endpoint": {"url": url}
      }
    }
  )
deploy.sh — Un comando para todo
#!/bin/bash
# Deploy completo de ShopMind a AWS

# 1. Deploy MCP servers
for s in price search reviews memory; do
  agentcore configure \
    -e "mcp_${s}_server.py" \
    --protocol MCP \
    --name "shopmind-${s}"
  agentcore launch
done

# 2. Crear Gateway + registrar targets
python infra/setup_gateway.py

# 3. Deploy supervisor agent
agentcore configure \
  -e agents/supervisor_agent.py \
  --name shopmind-supervisor
agentcore launch

# 4. Frontend → S3 + CloudFront
cd frontend && npm run build
aws s3 sync dist/ s3://shopmind-chile/

echo "🎉 ShopMind live!"
